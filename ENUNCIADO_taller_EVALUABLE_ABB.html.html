1. Cálculo de los estadísticos descriptivos

tabla_descriptiva <- listings0 %>% # Agrupamos por municipio y fecha (periodo) group_by(neighbourhood_cleansed, date) %>% # Calculamos los estadísticos para price y number_of_reviews summarise( N = n(), # Estadísticos para Precio mean_price = mean(price, na.rm = TRUE), sd_price = sd(price, na.rm = TRUE), min_price = min(price, na.rm = TRUE), max_price = max(price, na.rm = TRUE), # Estadísticos para Reviews mean_reviews = mean(number_of_reviews, na.rm = TRUE), sd_reviews = sd(number_of_reviews, na.rm = TRUE), min_reviews = min(number_of_reviews, na.rm = TRUE), max_reviews = max(number_of_reviews, na.rm = TRUE), .groups = “drop” # Desagrupar al finalizar )
